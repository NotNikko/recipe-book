category: "Dinner"
tags:
  - "Meat/ Oven"
prepTime: 30
cookTime: 45
servings: 6
difficulty: Easy
featured: false
favorite: false
ingredients:
  - "2lbs/900gr Ground beef"
  - "150gr Spinach"
  - "1 Medium yellow onion"
  - "3 Cloves garlic"
  - "50gr Ground parmigiano"
  - "2 eggs"
  - "100/200gr Italian grated bread"
  - "2tbsp Bbq sauce"
  - "1tsp Worchestershire sauce"
  - "1tsp Soy sauce"
  - "2tbsp EVOO"
  - "1/3 Cup Ketchup"
  - "2 Tbsp Brown sugar"
  - "Spices"
  - "Salt/Pepper/Cayenne/Paprica etc"
instructions:
  - "Add EVOO to pan"
  - "Chop onion & add to pan"
  - "Chop garlic & add to pan"
  - "Add 150gr of spinach & sauté"
  - "Flavor with salt & Pepper"
  - "Cook for 2/3 minutes"
  - "In bowl add & mix by hand"
  - "Ground beef & beaten eggs"
  - "Add bread crumbs & parmigiano"
  - "Add cooled down spinaches"
  - "Add soy sauce & Worchester"
  - "Add spices & seasonings"
  - "If too wet add more crumbs"
  - "Put in baking container & push evenly."
  - "Bake @ 400 for 45 minutes"
  - "Make toppings sauce"
  - "Mix ketchup with brown sugar"
  - "Pour over 5 minutes before removing from oven"
  - "Check internal temp 160F"
  - "Remove & rest for 5/10 minutes"
  - "Serve hot"
